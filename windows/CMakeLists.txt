cmake_minimum_required(VERSION 3.20)
project(CoincideWindowsCore LANGUAGES CXX)
enable_testing()
add_executable(CoincideWorkspaceTests Tests/WorkspaceTests.cpp)
target_compile_features(CoincideWorkspaceTests PRIVATE cxx_std_20)
if(MSVC)
  target_compile_options(CoincideWorkspaceTests PRIVATE /W4 /WX /utf-8)
  target_compile_definitions(CoincideWorkspaceTests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE)
else()
  target_compile_options(CoincideWorkspaceTests PRIVATE -Wall -Wextra -Werror -pedantic)
endif()
add_test(NAME workspace COMMAND CoincideWorkspaceTests)
